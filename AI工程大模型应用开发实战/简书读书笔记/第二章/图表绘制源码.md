# 图表绘制源码

下列Python代码生成本组八张PNG。需要Pillow及可用的中文字体；ROOT、字体和临时输出路径可按运行环境修改。

INK='#153647';MUTED='#556E7B';TEAL='#0C837F';BLUE='#276FA7';PURPLE='#7753A7'
def font(s,b=False):return ImageFont.truetype(FB if b else F,s)
def center(draw,xy,text,size=32,fill=INK,bold=False):
    draw.text(xy,text,font=font(size,bold),fill=fill,anchor='mm')
def base(title,sub,h=1000):
    im=Image.new('RGB',(1400,h),'#F5F8FA');d=ImageDraw.Draw(im)
    d.rectangle((0,0,1400,12),fill=TEAL)
    d.text((60,48),title,font=font(44,True),fill=INK)
    d.text((60,113),sub,font=font(25),fill=MUTED)
    d.line((60,h-84,1340,h-84),fill='#D6E0E5',width=2)
    d.text((60,h-60),'《AI工程》第二章读书笔记 · 自绘学习示意 · 2026-10-15',font=font(23),fill=MUTED)
    return im,d
def box(d,r,lines,fill='#E4F3F0',stroke=TEAL,size=32):
    d.rounded_rectangle(r,radius=19,fill=fill,outline=stroke,width=3)
    if isinstance(lines,str):lines=lines.split('\n')
    cx=(r[0]+r[2])/2;cy=(r[1]+r[3])/2
    for j,line in enumerate(lines):center(d,(cx,cy+(j-(len(lines)-1)/2)*(size+15)),line,size)
def arrow(d,a,b,color=TEAL):
    d.line([a,b],fill=color,width=5);ang=math.atan2(b[1]-a[1],b[0]-a[0]);ll=17
    pts=[b,(b[0]-ll*math.cos(ang-.5),b[1]-ll*math.sin(ang-.5)),(b[0]-ll*math.cos(ang+.5),b[1]-ll*math.sin(ang+.5))]
    d.polygon(pts,fill=color)
def save(im,name):im.save(OUT/name)

box(d,(80,520,490,720),'数据治理\n筛选、去重、版本与配比')

for i,(label,y1,y2) in enumerate(rows):
    box(d,(180,y1,1220,y2),label,fill='#E4F3F0' if i%2==0 else '#EAF0FA',stroke=TEAL if i%2==0 else BLUE,size=32)

im,d=base('稠密前馈与稀疏专家：计算走向哪里','专家是可训练模块；不默认对应人工定义的职业或知识门类。',1000)
center(d,(345,215),'稠密前馈',36,TEAL,True);center(d,(1050,215),'稀疏混合专家（MoE）',36,PURPLE,True)
box(d,(120,290,570,390),'当前token表示');box(d,(120,510,570,650),'共享的前馈模块')
arrow(d,(345,390),(345,510));center(d,(345,755),'该层使用相同模块',28,MUTED)
box(d,(830,290,1270,390),'当前token表示');box(d,(880,450,1220,540),'路由器',fill='#F0EAF7',stroke=PURPLE)
arrow(d,(1050,390),(1050,450))
for x,label,active in [(755,'专家1',True),(965,'专家2',False),(1175,'专家3',True)]:
    box(d,(x,650,x+175,755),label,fill='#E4F3F0' if active else '#EDF0F2',stroke=TEAL if active else '#ADB9BF',size=26)
    if active:arrow(d,(1050,540),(x+87,650),PURPLE)
center(d,(1050,825),'仅示意选择部分专家；实际还需组合输出',25,MUTED)
d.line((695,205,695,850),fill='#D6E0E5',width=2);save(im,'03-moe.png')

im,d=base('训练预算与应用预算：两本相关的账','规模估算帮助规划资源；是否值得使用，仍取决于任务结果。',1000)
box(d,(80,210,625,325),'训练约束',fill='#E4F3F0',size=36)
box(d,(775,210,1320,325),'应用约束',fill='#EAF0FA',stroke=BLUE,size=36)
box(d,(80,405,625,610),'参数规模 · 数据质量与数量\n架构 · 设备 · 有效算力',size=30)

im,d=base('温度如何改变候选概率','固定logits：A=1，B=2；按softmax公式计算，未调用模型。',1000)
x0,y0=155,780;x1,y1=1280,235
for p in [0,.25,.5,.75,1]:

    xx=x0+(t-.1)/1.9*(x1-x0);d.text((xx-20,y0+20),str(t),font=font(25),fill=MUTED)
for isb,col in [(True,TEAL),(False,PURPLE)]:

center(d,(1150,290),'B（logit=2）',28,TEAL,True)
center(d,(1150,610),'A（logit=1）',28,PURPLE,True)
d.text((65,190),'概率',font=font(27),fill=INK);center(d,(730,855),'温度 T（大于0）',28)
